class Person {
    id: number;
    name: string;
    salary: number;

    constructor(id: number, name: string, salary: number) {
        this.id = id;
        this.name = name;
        this.salary = salary;
    }

    toString(): string {
        return `Person{id=${this.id}, name=${this.name}, salary=${this.salary}}`;
    }
}
